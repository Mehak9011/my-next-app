# ============================================================
# Headless WordPress — serve the public frontend from Next.js
# ------------------------------------------------------------
# Upload this file to the WordPress web root as:  .htaccess
# (Hostinger hPanel -> File Manager -> public_html -> .htaccess).
# REPLACE the whole file content with ONE of the variants below
# (then test https://cms.codexmattrix.com/about/).
#
# WHAT IT DOES
#   • WordPress backend / API stays on WordPress:
#       /wp-admin, /wp-login.php, /wp-json, /graphql, /wp-content,
#       /wp-includes, /wp-cron.php, /xmlrpc.php, /.well-known/
#   • EVERY OTHER URL (/, /about/, /services/, /_next/static/...)
#     is rendered by the Next.js site on Vercel while the browser
#     KEEPS the cms.codexmattrix.com/... URL.
# ============================================================

# =-=-=-=-=-=-=-=-=-=-=-=-=-=-=-=-=-=-=-=-=-=-=-=-=-=-=-=-=-=-=
# VARIANT 1 (RECOMMENDED — works on Hostinger without mod_proxy)
# PHP reverse proxy -> requires wordpress/proxy.php uploaded too.
# =-=-=-=-=-=-=-=-=-=-=-=-=-=-=-=-=-=-=-=-=-=-=-=-=-=-=-=-=-=-=
<IfModule mod_rewrite.c>
RewriteEngine On
RewriteBase /

# --- Never proxy WordPress internals or the proxy script itself ---
RewriteCond %{REQUEST_URI} !^/(proxy\.php|index\.php|wp-admin/?|wp-login\.php|wp-json|graphql|wp-content/|wp-includes/|wp-cron\.php|xmlrpc\.php|\.well-known/|readme\.html|license\.txt)
RewriteCond %{REQUEST_FILENAME} !-f
RewriteRule ^(.*)$ /proxy.php?_cmx=1 [L,QSA]

# --- Standard WordPress front controller (admin/API handling) ---
RewriteRule ^index\.php$ - [L]
RewriteCond %{REQUEST_FILENAME} !-f
RewriteRule . /index.php [L]
</IfModule>
